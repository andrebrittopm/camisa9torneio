[3. SNIFFING DE MAGIC BYTES (REC32)]
Implementação de autoridade final sobre MIME ignorando cabeçalhos do cliente.

Trecho (src/routes/api/public/av-payment-receipt.ts):
const arrayBuffer = await file.arrayBuffer();
const bytes = new Uint8Array(arrayBuffer.slice(0, 8));
let detectedMime = "";
if (bytes[0] === 0xFF && bytes[1] === 0xD8 && bytes[2] === 0xFF) { detectedMime = "image/jpeg"; }
else if (bytes[0] === 0x89 && bytes[1] === 0x50 && bytes[2] === 0x4E && bytes[3] === 0x47 ...) { detectedMime = "image/png"; }
else if (bytes[0] === 0x25 && bytes[1] === 0x50 && bytes[2] === 0x44 && bytes[3] === 0x46 ...) { detectedMime = "application/pdf"; }
